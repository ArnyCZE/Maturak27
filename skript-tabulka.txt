/*
 * Maturák 2027 – prodej vstupenek přes pokladní, 5min rezervace, QR platby a jejich záznam
 * (verze 5 – jeden nákup = místa u více stolů, jméno a příjmení kupujícího)
 *
 * Jak to funguje:
 *  - Plánek s obsazeností na webu vidí všichni.
 *  - Místa rezervovat a generovat QR kód smí jen přihlášená pokladní (list „Pokladní“).
 *  - Pokladní vybere místa (klidně u více stolů), vyplní jméno a příjmení kupujícího a vygeneruje
 *    jeden QR kód na celý nákup → místa jsou 5 minut „drženo“.
 *  - Když kupující zaplatí, pokladní klikne „Zaplaceno“ → místa jsou natrvalo „zaplaceno“.
 *    Když pokladní nákup zruší nebo do 5 minut nepotvrdí, místa se uvolní.
 *  - V listu „Prodej“ je za každý stůl nákupu jeden řádek (se stejným VS), v listu „QR kódy“
 *    je za každý QR kód jeden řádek (kdo, komu, kdy, stoly, místa, VS, stav).
 *  - Skript každých 5 minut projde e-maily od Fio o příchozích platbách a k zaplaceným nákupům
 *    doplní „Ověřeno bankou“ – podle variabilního symbolu (číslo nákupu 27xxxxx).
 *
 * Instalace / aktualizace:
 *   1. V Apps Scriptu smaž starý kód, vlož tenhle → Uložit.
 *   2. V seznamu funkcí nahoře vyber „setup“ → Spustit.
 *   3. Implementovat → Spravovat implementace → tužka → Verze: „Nová verze“ → Implementovat.
 */

var SPREADSHEET_ID = "1O-i3_C3p2RyKldJzpj98r_GLJsaVL6VH0yNoLWV0-v0";   // Maturák 2027 – vstupenky (soukromé)
var PRICE = 450;                       // cena jedné vstupenky v Kč
var HOLD_MINUTES = 5;                  // jak dlouho QR kód drží místa
var SESSION_HOURS = 6;                 // jak dlouho platí přihlášení pokladní
var ACCOUNT_NUMBER = "2203629553";     // účet Plesy 4FH (Fio)
var PAID_LABEL = "ples-zpracovano";    // štítek v Gmailu pro už zpracované platby
var HALL_BY_DIGIT = { "1": "VS", "2": "LS", "3": "PS", "4": "G" };

// sloupce v listu Prodej (1 = A)
var COL = { name: 1, contact: 2, table: 3, count: 4, status: 5, note: 6, vs: 7, amount: 8, created: 9, paid: 10, seats: 11, bank: 12 };
// sloupce v listu QR kódy
var LOG = { created: 1, cashier: 2, buyer: 3, table: 4, seats: 5, count: 6, amount: 7, vs: 8, status: 9, confirmed: 10, bank: 11, payer: 12, token: 13 };
var LOG_HEADER = ["Vygenerováno", "Pokladní", "Kupující", "Stoly", "Místa", "Počet", "Částka (Kč)", "VS", "Stav",
                  "Potvrzeno pokladní", "Ověřeno bankou (Fio)", "Plátce (z banky)", "Kód"];
var CASHIER_HEADER = ["Jméno", "Přihlašovací jméno", "Heslo", "Aktivní (ano/ne)"];
var INACTIVE = { "vypršelo": 1, "zrušeno": 1 };

/* ---------- rozhraní pro web (JSONP přes GET, funguje i z GitHub Pages) ---------- */

function doGet(e) {
  var p = (e && e.parameter) || {};
  var result;
  try {
    if (p.action === "state") result = { ok: true, tables: state_() };
    else if (p.action === "login") result = login_(p.login, p.password);
    else if (p.action === "hold" || p.action === "release" || p.action === "confirm" || p.action === "check") {
      var cashier = cashierFromSession_(p.session);
      if (!cashier) result = { ok: false, error: "Přihlášení vypršelo – přihlas se prosím znovu.", loggedOut: true };
      else if (p.action === "check") result = { ok: true, name: cashier };
      else result = withLock_(function () {
        if (p.action === "hold") return hold_(p.items, p.firstName, p.lastName, cashier);
        if (p.action === "confirm") return confirm_(p.vs, p.token);
        return release_(p.vs, p.token);
      });
    }
    else result = { ok: true, info: "Maturák 2027 – rezervace běží (verze 5)." };
  } catch (err) {
    result = { ok: false, error: "Chyba skriptu: " + err.message };
  }
  var json = JSON.stringify(result);
  if (p.callback && /^[\w.]+$/.test(p.callback)) {
    return ContentService.createTextOutput(p.callback + "(" + json + ")").setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}

/* Zápisy probíhají po jednom, aby dva lidé nezabrali stejné místo. */
function withLock_(fn) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) return { ok: false, error: "Systém je teď vytížený, zkus to prosím za pár vteřin." };
  try { return fn(); } finally { lock.releaseLock(); }
}

/* ---------- přihlášení pokladních ---------- */

function login_(login, password) {
  login = String(login || "").trim().toLowerCase();
  var cache = CacheService.getScriptCache();
  var failKey = "fail_" + login;
  var fails = parseInt(cache.get(failKey) || "0", 10);
  if (fails >= 5) return { ok: false, error: "Příliš mnoho pokusů – zkus to za 10 minut." };

  var rows = sheet_("Pokladní").getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    var active = String(r[3] || "").trim().toLowerCase();
    if (login && String(r[1] || "").trim().toLowerCase() === login &&
        String(r[2]) !== "" && String(r[2]) === String(password || "") && active !== "ne") {
      var session = Utilities.getUuid().replace(/-/g, "");
      cache.put("s_" + session, String(r[0] || login), SESSION_HOURS * 3600);
      cache.remove(failKey);
      return { ok: true, session: session, name: String(r[0] || login) };
    }
  }
  cache.put(failKey, String(fails + 1), 600);
  return { ok: false, error: "Špatné přihlašovací jméno nebo heslo." };
}

function cashierFromSession_(session) {
  if (!session) return null;
  return CacheService.getScriptCache().get("s_" + session);
}

/* ---------- obsazenost a rezervace ---------- */

/* Kapacity stolů (z listu Stoly, na 10 minut v mezipaměti – čtení je pak rychlé). */
function capacities_() {
  var cache = CacheService.getScriptCache();
  var cached = cache.get("caps");
  if (cached) return JSON.parse(cached);
  var rows = sheet_("Stoly").getRange("A2:C200").getValues(), caps = {};
  rows.forEach(function (r) { if (r[0]) caps[r[0]] = Number(r[2]) || 0; });
  cache.put("caps", JSON.stringify(caps), 600);
  return caps;
}

/* Je řádek aktivní? Podržení starší než HOLD_MINUTES se bere jako uvolněné, i když ho časovač ještě nepřepsal. */
function isActive_(r, now) {
  var st = r[COL.status - 1];
  if (INACTIVE[st]) return false;
  if (st === "drženo") {
    var created = r[COL.created - 1];
    if (created instanceof Date && created.getTime() < now - HOLD_MINUTES * 60 * 1000) return false;
  }
  return true;
}

/* Kapacita a obsazené (zaplacené, rezervované i právě držené) židle u každého stolu. */
function state_() {
  var caps = capacities_(), out = {}, now = Date.now();
  Object.keys(caps).forEach(function (id) { out[id] = { cap: caps[id], seats: [] }; });
  var data = sheet_("Prodej").getDataRange().getValues();
  for (var j = 1; j < data.length; j++) {
    var r = data[j], t = out[r[COL.table - 1]];
    if (!t || !isActive_(r, now)) continue;
    var seats = parseSeats_(r[COL.seats - 1]);
    // ručně zapsané prodeje bez čísel židlí obsadí nejnižší volná místa
    for (var s = 1; seats.length < Number(r[COL.count - 1]) && s <= t.cap; s++) {
      if (t.seats.indexOf(s) === -1 && seats.indexOf(s) === -1) seats.push(s);
    }
    t.seats = t.seats.concat(seats);
  }
  return out;
}

/* items = "VS 12:1,2;LS 50:3" – místa u jednoho nebo více stolů, jeden QR kód na celý nákup. */
function hold_(itemsParam, firstName, lastName, cashier) {
  firstName = String(firstName || "").trim();
  lastName = String(lastName || "").trim();
  if (!firstName || !lastName) return { ok: false, error: "Vyplň jméno i příjmení kupujícího." };
  var buyer = firstName + " " + lastName;

  var items = parseItems_(itemsParam);
  if (!items.length) return { ok: false, error: "Vyber aspoň jedno místo." };
  var tables = state_(), count = 0;
  for (var i = 0; i < items.length; i++) {
    var it = items[i], t = tables[it.table];
    if (!t || !t.cap) return { ok: false, error: "Stůl " + it.table + " neexistuje." };
    if (it.seats.some(function (s) { return s < 1 || s > t.cap; })) return { ok: false, error: "Neplatné číslo místa u stolu " + it.table + "." };
    var clash = it.seats.filter(function (s) { return t.seats.indexOf(s) !== -1; });
    if (clash.length) return { ok: false, error: "Stůl " + it.table + ": místa " + clash.join(", ") + " si právě někdo zabral. Vyber jiná.", tables: tables };
    count += it.seats.length;
  }

  var vs = nextOrderVs_();
  var token = Utilities.getUuid().slice(0, 8);
  var now = new Date(), amount = count * PRICE;
  var prodej = sheet_("Prodej");
  items.forEach(function (it) {
    prodej.appendRow([buyer, "pokladní: " + cashier, it.table, it.seats.length, "drženo", "web " + token, vs,
                      it.seats.length * PRICE, now, "", "'" + it.seats.join(","), ""]);
  });
  sheet_("QR kódy").appendRow([now, cashier, buyer,
    items.map(function (it) { return it.table; }).join(", "),
    items.map(function (it) { return it.table + ": " + it.seats.join(","); }).join(" · "),
    count, amount, vs, "drženo", "", "", "", token]);
  return { ok: true, vs: vs, token: token, amount: amount, expiresAt: now.getTime() + HOLD_MINUTES * 60 * 1000 };
}

/* Číslo nákupu = variabilní symbol: 27 + pořadové číslo (2700001, 2700002 …). */
function nextOrderVs_() {
  var props = PropertiesService.getScriptProperties();
  var n = parseInt(props.getProperty("orderSeq") || "0", 10) + 1;
  props.setProperty("orderSeq", String(n));
  return "27" + ("0000" + n).slice(-5);
}

/* Všechny řádky nákupu (každý stůl jeden řádek) podle VS a kódu. */
function findOrder_(vs, token) {
  var sh = sheet_("Prodej"), data = sh.getDataRange().getValues(), rows = [];
  for (var i = 1; i < data.length; i++) {
    var r = data[i];
    if (String(r[COL.vs - 1]) === String(vs) && String(r[COL.note - 1]).indexOf("web " + token) === 0) {
      rows.push({ row: i + 1, values: r });
    }
  }
  return { sheet: sh, rows: rows };
}

/* Místa z nákupu, která mezitím zabral někdo jiný (jen u propadlého podržení). */
function orderClashes_(rows) {
  var tables = state_(), now = Date.now(), msgs = [];
  rows.forEach(function (h) {
    if (h.values[COL.status - 1] === "zaplaceno" || isActive_(h.values, now)) return;
    var table = h.values[COL.table - 1], t = tables[table] || { seats: [] };
    var clash = parseSeats_(h.values[COL.seats - 1]).filter(function (s) { return t.seats.indexOf(s) !== -1; });
    if (clash.length) msgs.push(table + ": " + clash.join(", "));
  });
  return msgs;
}

/* Pokladní potvrdil(a), že kupující zaplatil → místa jsou prodaná natrvalo. */
function confirm_(vs, token) {
  var o = findOrder_(vs, token);
  if (!o.rows.length) return { ok: false, error: "Rezervace nenalezena." };
  var clash = orderClashes_(o.rows);
  if (clash.length) return { ok: false, error: "Rezervace vypršela a tato místa už má někdo jiný – " + clash.join("; ") + "." };
  var now = new Date();
  o.rows.forEach(function (h) {
    if (h.values[COL.status - 1] === "zaplaceno") return;
    o.sheet.getRange(h.row, COL.status).setValue("zaplaceno");
    o.sheet.getRange(h.row, COL.paid).setValue(now);
  });
  updateLog_(token, { status: "zaplaceno", confirmed: now });
  return { ok: true };
}

/* Pokladní nákup zrušil(a) → místa se hned uvolní. */
function release_(vs, token) {
  var o = findOrder_(vs, token);
  if (!o.rows.length) return { ok: false, error: "Rezervace nenalezena." };
  var changed = false;
  o.rows.forEach(function (h) {
    if (h.values[COL.status - 1] !== "drženo") return;
    o.sheet.getRange(h.row, COL.status).setValue("zrušeno");
    changed = true;
  });
  if (changed) updateLog_(token, { status: "zrušeno" });
  return { ok: true };
}

/* ---------- automatika (časovače z funkce setup) ---------- */

/* Přepíše propadlá podržení na „vypršelo“ (obsazenost je počítá jako volná už dřív – viz isActive_). */
function expireHolds() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return;
  try {
    var sh = sheet_("Prodej"), data = sh.getDataRange().getValues(), now = Date.now(), done = {};
    for (var i = 1; i < data.length; i++) {
      if (data[i][COL.status - 1] === "drženo" && !isActive_(data[i], now)) {
        sh.getRange(i + 1, COL.status).setValue("vypršelo");
        var note = String(data[i][COL.note - 1] || "");
        if (note.indexOf("web ") === 0 && !done[note]) { done[note] = 1; updateLog_(note.slice(4, 12), { status: "vypršelo" }); }
      }
    }
  } finally {
    lock.releaseLock();
  }
}

/* Projde e-maily od Fio o příchozích platbách na plesový účet a doplní „Ověřeno bankou“. */
function checkPayments() {
  var threads = GmailApp.search("from:fio.cz newer_than:30d -label:" + PAID_LABEL, 0, 50);
  if (!threads.length) return;
  var label = GmailApp.getUserLabelByName(PAID_LABEL) || GmailApp.createLabel(PAID_LABEL);

  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return;           // zkusí to znovu za 5 minut
  try {
    var sh = sheet_("Prodej");
    threads.forEach(function (thread) {
      thread.getMessages().forEach(function (msg) { processPayment_(sh, msg); });
      thread.addLabel(label);
    });
  } finally {
    lock.releaseLock();
  }
}

function processPayment_(sh, msg) {
  var body = msg.getPlainBody();
  if (body.indexOf(ACCOUNT_NUMBER) === -1) return;                    // platba na jiný účet

  var vsMatch = body.match(/(?:VS|variabiln\S* symbol)\s*:?\s*(\d{6,7})\b/i);
  if (!vsMatch) return;
  var vs = vsMatch[1];

  var amountMatch = body.match(/(?:částka|objem)\s*:?\s*([\d\s .]+,?\d*)/i);
  var amount = amountMatch ? parseFloat(amountMatch[1].replace(/[\s .]/g, "").replace(",", ".")) : NaN;
  var payerMatch = body.match(/(?:název protiúčtu|protiúčet|plátce)\s*:?\s*([^\n\r]+)/i);
  var payer = payerMatch ? payerMatch[1].trim() : "";

  // nejnovější nákup s tímto VS, který ještě nemá ověření (všechny jeho řádky = stoly)
  var data = sh.getDataRange().getValues(), rows = [], note = null;
  for (var i = data.length - 1; i >= 1; i--) {
    if (String(data[i][COL.vs - 1]) !== vs || data[i][COL.bank - 1]) continue;
    var n = String(data[i][COL.note - 1] || "");
    if (note === null) note = n;
    if (n === note) rows.push({ row: i + 1, values: data[i] });
  }
  var expected = rows.reduce(function (sum, h) { return sum + Number(h.values[COL.amount - 1] || 0); }, 0);
  var bankNote = "ano · " + Utilities.formatDate(msg.getDate(), "Europe/Prague", "d.M.yyyy HH:mm") +
                 (isNaN(amount) ? " (částku zkontroluj)" : " · " + amount + " Kč" + (expected && amount < expected ? " – MÁLO, má být " + expected + " Kč" : "")) +
                 (payer ? " · " + payer : "");

  if (!rows.length) {
    var old = decodeVs_(vs);      // starý formát VS (stůl + místa), jinak neznámá platba
    sh.appendRow([payer, "", old ? old.table : "", old ? old.seats.length : 0, old ? "zaplaceno" : "zrušeno",
                  old ? "platba bez QR z webu – zkontroluj" : "neznámý VS – zkontroluj platbu", vs,
                  isNaN(amount) ? "" : amount, msg.getDate(), "", old ? "'" + old.seats.join(",") : "", bankNote]);
    return;
  }

  rows.forEach(function (h) { sh.getRange(h.row, COL.bank).setValue(bankNote); });
  // peníze přišly, ale pokladní nepotvrdil(a) – zapíšeme jako zaplacené, pokud místa nikdo jiný nemá
  var unpaid = rows.filter(function (h) { return h.values[COL.status - 1] !== "zaplaceno"; });
  if (unpaid.length) {
    var clash = orderClashes_(unpaid);
    unpaid.forEach(function (h) { sh.getRange(h.row, COL.status).setValue(clash.length ? "zrušeno" : "zaplaceno"); });
    if (clash.length) sh.getRange(unpaid[0].row, COL.note).setValue(note + " KONFLIKT: místa " + clash.join("; ") + " už prodaná – vrátit peníze.");
  }
  if (note.indexOf("web ") === 0) updateLog_(note.slice(4, 12), { bank: bankNote, payer: payer });
}

/* Spusť jednou ručně: vytvoří chybějící listy, nastaví časové pásmo, časovače a štítek v Gmailu. */
function setup() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  ss.setSpreadsheetTimeZone("Europe/Prague");
  ensureSheet_("Pokladní", CASHIER_HEADER);
  ensureSheet_("QR kódy", LOG_HEADER);
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger("expireHolds").timeBased().everyMinutes(1).create();
  ScriptApp.newTrigger("checkPayments").timeBased().everyMinutes(5).create();
  if (!GmailApp.getUserLabelByName(PAID_LABEL)) GmailApp.createLabel(PAID_LABEL);
  CacheService.getScriptCache().remove("caps");
}

/* ---------- pomocné funkce ---------- */

/* Upraví řádek v listu QR kódy podle kódu (tokenu) nákupu. */
function updateLog_(token, fields) {
  var sh = sheet_("QR kódy"), data = sh.getDataRange().getValues();
  for (var i = data.length - 1; i >= 1; i--) {
    if (String(data[i][LOG.token - 1]) !== String(token)) continue;
    if (fields.status) sh.getRange(i + 1, LOG.status).setValue(fields.status);
    if (fields.confirmed) sh.getRange(i + 1, LOG.confirmed).setValue(fields.confirmed);
    if (fields.bank) sh.getRange(i + 1, LOG.bank).setValue(fields.bank);
    if (fields.payer) sh.getRange(i + 1, LOG.payer).setValue(fields.payer);
    return;
  }
}

function ensureSheet_(name, header) {
  var ss = spreadsheet_ || (spreadsheet_ = SpreadsheetApp.openById(SPREADSHEET_ID));
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.getRange(1, 1, 1, header.length).setValues([header]).setFontWeight("bold").setBackground("#f2d98c");
    sh.setFrozenRows(1);
  }
  return sh;
}

/* "VS 12:1,2;LS 50:3" → [{ table: "VS 12", seats: [1,2] }, { table: "LS 50", seats: [3] }] */
function parseItems_(v) {
  var seen = {};
  return String(v || "").split(";").map(function (part) {
    var i = part.indexOf(":");
    if (i === -1) return null;
    var table = part.slice(0, i).trim(), seats = parseSeats_(part.slice(i + 1));
    seats = seats.filter(function (s, k) { return seats.indexOf(s) === k; });
    if (!table || !seats.length || seen[table]) return null;
    seen[table] = 1;
    return { table: table, seats: seats };
  }).filter(Boolean);
}

/* Starý formát VS (do verze 4): číslice sálu + 2 číslice stolu + 3 číslice „masky“ míst. */
function decodeVs_(vs) {
  var m = String(vs).match(/^([1-4])(\d{2})(\d{3})$/);
  if (!m) return null;
  var mask = parseInt(m[3], 10), seats = [];
  for (var i = 0; i < 8; i++) if (mask & (1 << i)) seats.push(i + 1);
  if (!seats.length) return null;
  return { table: HALL_BY_DIGIT[m[1]] + " " + parseInt(m[2], 10), seats: seats };
}

function parseSeats_(v) {
  return String(v || "").replace(/^'/, "").split(",").map(function (s) { return parseInt(s, 10); }).filter(function (n) { return n > 0; });
}

var spreadsheet_ = null;
function sheet_(name) {
  spreadsheet_ = spreadsheet_ || SpreadsheetApp.openById(SPREADSHEET_ID);
  var sh = spreadsheet_.getSheetByName(name);
  if (!sh) {
    if (name === "Pokladní") return ensureSheet_(name, CASHIER_HEADER);
    if (name === "QR kódy") return ensureSheet_(name, LOG_HEADER);
    throw new Error("V tabulce chybí list „" + name + "“.");
  }
  return sh;
}
