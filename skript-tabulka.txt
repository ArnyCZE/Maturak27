/*
 * Maturák 2027 – automatické zapisování zaplacených vstupenek
 *
 * Jak to funguje: na webu si kupující vybere místa u stolu a zaplatí QR kódem.
 * Variabilní symbol v QR kódu říká, který stůl a která místa platí.
 * Tenhle skript každých 5 minut projde e-maily od Fio o příchozích platbách
 * a zaplacená místa zapíše do listu Prodej – web je pak ukáže jako obsazená.
 *
 * Instalace (jednou):
 *   Rozšíření → Apps Script → smaž, co tam je, a vlož tohle → Uložit
 *   → nahoře vyber funkci „setup“ → Spustit → povolit přístup.
 *   Implementace (webová aplikace) ani PIN už nejsou potřeba.
 */

var PRICE = 450;                       // cena jedné vstupenky v Kč
var ACCOUNT_NUMBER = "2203629553";     // účet Plesy 4FH (Fio)
var PAID_LABEL = "ples-zpracovano";    // štítek v Gmailu pro už zpracované platby
var HALL_BY_DIGIT = { "1": "VS", "2": "LS", "3": "PS", "4": "G" };

// sloupce v listu Prodej (1 = A)
var COL = { name: 1, contact: 2, table: 3, count: 4, status: 5, note: 6, vs: 7, amount: 8, created: 9, paid: 10, seats: 11 };
var INACTIVE = { "vypršelo": 1, "zrušeno": 1 };

/* VS = číslice sálu + 2 číslice stolu + 3 číslice „masky“ míst (bit 1 = místo 1 …), např. 112005 = VS 12, místa 1 a 3. */
function decodeVs_(vs) {
  var m = String(vs).match(/^([1-4])(\d{2})(\d{3})$/);
  if (!m) return null;
  var mask = parseInt(m[3], 10), seats = [];
  for (var i = 0; i < 8; i++) if (mask & (1 << i)) seats.push(i + 1);
  if (!seats.length) return null;
  return { table: HALL_BY_DIGIT[m[1]] + " " + parseInt(m[2], 10), seats: seats };
}

/* Projde e-maily od Fio o příchozích platbách na plesový účet a zapíše zaplacená místa. */
function checkPayments() {
  var label = GmailApp.getUserLabelByName(PAID_LABEL) || GmailApp.createLabel(PAID_LABEL);
  var threads = GmailApp.search("from:fio.cz newer_than:30d -label:" + PAID_LABEL);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Prodej");

  threads.forEach(function (thread) {
    thread.getMessages().forEach(function (msg) {
      var body = msg.getPlainBody();
      if (body.indexOf(ACCOUNT_NUMBER) === -1) return;                      // platba na jiný účet

      var vsMatch = body.match(/(?:VS|variabiln\S* symbol)\s*:?\s*(\d{6})\b/i);
      if (!vsMatch) return;
      var order = decodeVs_(vsMatch[1]);
      if (!order) return;
      if (alreadyRecorded_(sheet, vsMatch[1], msg.getDate())) return;

      var amountMatch = body.match(/(?:částka|objem)\s*:?\s*([\d\s .]+,?\d*)/i);
      var amount = amountMatch ? parseFloat(amountMatch[1].replace(/[\s .]/g, "").replace(",", ".")) : NaN;
      var payerMatch = body.match(/(?:název protiúčtu|protiúčet|plátce|od)\s*:?\s*([^\n\r]+)/i);
      var noteMatch = body.match(/(?:zpráva pro příjemce|zpráva)\s*:?\s*([^\n\r]+)/i);

      var expected = order.seats.length * PRICE;
      var notes = [];
      if (isNaN(amount)) notes.push("POZOR: nepodařilo se přečíst částku – zkontroluj.");
      else if (amount < expected) notes.push("POZOR: zaplaceno " + amount + " Kč místo " + expected + " Kč.");
      var clash = takenSeats_(sheet, order.table).filter(function (s) { return order.seats.indexOf(s) !== -1; });
      if (clash.length) notes.push("KONFLIKT: místa " + clash.join(",") + " už někdo zaplatil – vrátit peníze nebo nabídnout jiná místa.");
      if (noteMatch) notes.push("Zpráva: " + noteMatch[1].trim());

      sheet.appendRow([
        payerMatch ? payerMatch[1].trim() : "", "", order.table, order.seats.length,
        clash.length ? "zrušeno" : "zaplaceno", notes.join(" "), vsMatch[1],
        isNaN(amount) ? "" : amount, msg.getDate(), msg.getDate(), order.seats.join(",")
      ]);
    });
    thread.addLabel(label);
  });
}

/* Místa u stolu, která už jsou obsazená (zaplacená nebo zapsaná ručně). */
function takenSeats_(sheet, table) {
  var data = sheet.getDataRange().getValues(), seats = [];
  for (var i = 1; i < data.length; i++) {
    var r = data[i];
    if (r[COL.table - 1] !== table || INACTIVE[r[COL.status - 1]]) continue;
    String(r[COL.seats - 1] || "").split(",").forEach(function (s) { if (s.trim()) seats.push(parseInt(s, 10)); });
  }
  return seats;
}

/* Ochrana proti dvojímu zápisu stejné platby. */
function alreadyRecorded_(sheet, vs, date) {
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    var r = data[i], paid = r[COL.paid - 1];
    if (String(r[COL.vs - 1]) === String(vs) && paid instanceof Date && paid.getTime() === date.getTime()) return true;
  }
  return false;
}

/* Spusť jednou ručně: nastaví časovač (každých 5 minut) a štítek v Gmailu. */
function setup() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger("checkPayments").timeBased().everyMinutes(5).create();
  if (!GmailApp.getUserLabelByName(PAID_LABEL)) GmailApp.createLabel(PAID_LABEL);
}
